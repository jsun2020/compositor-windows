cmake_minimum_required(VERSION 3.21)
project(CompositorHeif LANGUAGES CXX)
find_package(libheif CONFIG REQUIRED)
add_executable(CompositorHeif heif-decode.cpp)
target_compile_features(CompositorHeif PRIVATE cxx_std_17)
target_link_libraries(CompositorHeif PRIVATE heif)
if(MSVC)
  target_compile_options(CompositorHeif PRIVATE /W4 /utf-8)
endif()
install(TARGETS CompositorHeif RUNTIME DESTINATION bin)
